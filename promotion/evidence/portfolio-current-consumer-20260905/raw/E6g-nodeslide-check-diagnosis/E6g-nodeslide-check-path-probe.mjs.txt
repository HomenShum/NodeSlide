function child(cwd, args) {
  const r = spawnSync(process.execPath, args, {cwd, encoding:'utf8'});
  return {cwd, args, status:r.status, stdout:r.stdout, stderr:r.stderr, error:r.error ? {code:r.error.code, syscall:r.error.syscall, message:r.error.message}:null};
}
function manifests(root) {
  try {
    const paths = readdirSync(join(root,'packages'),{withFileTypes:true}).filter(e=>e.isDirectory()).map(e=>join(root,'packages',e.name,'package.json')).concat(join(root,'mcp/package.json'));
    const values = paths.map(p=>JSON.parse(readFileSync(p,'utf8')));
    const convex = values.find(v=>v.name==='@nodeslide/convex');
    return {count:values.length, offenders:values.filter(v=>/\bnpm\s+run\b/.test(v.scripts?.build??'')).map(v=>v.name), convexBothOutputs:/\btsup\b/.test(convex.scripts.build) && /tsc -p tsconfig\.component\.json/.test(convex.scripts.build)};
  } catch(error) { return {error:{code:error.code,message:error.message}}; }
}

function discover(root) {
  let entries;
  try { entries=readdirSync(root,{recursive:true,withFileTypes:true}); }
  catch(error) { if(error.code==='ENOENT' && !existsSync(root)) return []; throw error; }
  return entries.filter(e=>e.isFile() && e.name==='trace.zip').map(e=>join(e.parentPath,e.name)).sort();
}
