def sha(b): return hashlib.sha256(b).hexdigest()
def readj(p): return json.loads(p.read_text(encoding='utf-8-sig'))
def writej(p, x): p.write_text(json.dumps(x, ensure_ascii=False, indent=2) + '\n', encoding='utf-8', newline='\n')
def git(*args): return subprocess.check_output(['git', '--no-optional-locks', '-C', str(R), *args])
def snap():
    return {'head': git('rev-parse', 'HEAD').decode().strip(),
            'indexEntriesSha256': sha(git('ls-files', '--stage', '-z')),
            'refsSha256': sha(git('show-ref'))}

changes = []
for f in freeze['source'][2:]:
    path = f['path']
    old = (P / 'E6g-nodeslide-multiline-citation-before' / path).read_bytes()
    new = (R / path).read_bytes()
    if path.endswith('.tour'):
        oj, nj = json.loads(old), json.loads(new)
        allowed = (255, 256) if path.startswith('.tours/01') else (298, 299)
        changed = []
        for i, (o, n) in enumerate(zip(oj['steps'], nj['steps'])):
            if o != n:
                assert o['line'] == allowed[0] and n['line'] == allowed[1]
                assert o['file'] == 'convex/lib/nodeslideValidators.ts'
                oc = dict(o); oc['line'] = n['line']
                assert oc == n
                changed.append(i)
        assert len(changed) == 1
        oc = json.loads(old); oc['steps'][changed[0]]['line'] = allowed[1]
        assert oc == nj
        assert old.replace(('"line": ' + str(allowed[0])).encode(), ('"line": ' + str(allowed[1])).encode()) == new
        changes.append({'path': path, 'only': f'validator anchor {allowed[0]} -> {allowed[1]}', 'step': changed[0], 'exactOtherBytes': True})
    else:
        token = b'nodeslideValidators.ts:255'
        # Markdown file citations use either colon syntax or #L syntax.
        if token not in old: token = b'nodeslideValidators.ts#L255'
        assert old.count(token) >= 1, path
        assert old.replace(token, token.replace(b'255', b'256')) == new, path
        changes.append({'path': path, 'only': 'validator anchor 255 -> 256', 'occurrences': old.count(token), 'exactOtherBytes': True})
    (O / ('citation-' + Path(path).name + '.diff')).write_text(''.join(difflib.unified_diff(old.decode('utf-8').splitlines(True), new.decode('utf-8').splitlines(True), fromfile='before/' + path, tofile='current/' + path)), encoding='utf-8', newline='\n')
