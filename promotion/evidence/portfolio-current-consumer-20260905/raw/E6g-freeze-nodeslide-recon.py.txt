commands = []
def git(repo, *args):
    argv = ['git', '--no-optional-locks', '-C', str(repo), *args]
    r = subprocess.run(argv, capture_output=True, timeout=40)
    commands.append({'argv': argv, 'exitCode': r.returncode, 'stdoutSha256': sha(r.stdout), 'stderr': r.stderr.decode('utf-8', 'replace')})
    if r.returncode: raise RuntimeError(f'Git read failed: {args!r}: {r.returncode}')
    return r.stdout

for saved in before:
    repo = saved['repo']
    initial = next(x for x in discovery if x['repo'] == repo)
    head = git(repo, 'rev-parse', 'HEAD').decode().strip()
    refs = git(repo, 'for-each-ref', '--format=%(refname) %(objectname) %(upstream:short)')
    registration = git(repo, 'worktree', 'list', '--porcelain')
    check(f'{repo}: HEAD unchanged', head == saved['head'])
    check(f'{repo}: exact refs unchanged', refs.decode() == initial['refs']['stdout'])
    check(f'{repo}: exact worktree registration unchanged', registration.decode() == initial['worktrees']['stdout'])
    entry = {'repo': repo, 'head': head, 'refsSha256': sha(refs), 'worktreeRegistrationSha256': sha(registration), 'worktrees': []}
    for wt in saved['worktrees']:
        path = wt['worktree']
        index = git(path, 'ls-files', '--stage', '-z')
        status = git(path, 'status', '--porcelain=v1', '-z', '--untracked-files=all')
        matched_args = ['status', '--porcelain=v1', '-z', '--untracked-files=all']
        if sha(status) != wt['statusSha256']:
            # Retain alternate read serialization rather than pretending an unequal digest matches.
            plain = git(path, 'status', '--porcelain=v1', '--untracked-files=all')
            if sha(plain) == wt['statusSha256']:
                status, matched_args = plain, ['status', '--porcelain=v1', '--untracked-files=all']
        check(f'{path}: index bytes unchanged', sha(index) == wt['indexSha256'])
        check(f'{path}: status metadata unchanged', sha(status) == wt['statusSha256'])
        entry['worktrees'].append({'worktree': path, 'indexSha256': sha(index), 'statusSha256': sha(status), 'statusArgs': matched_args})
    remote = git(repo, 'ls-remote', '--symref', 'origin', 'HEAD', 'refs/heads/main').decode()
    entry['remoteMain'] = remote
    check(f'{repo}: remote main still frozen commit', f'{H}\trefs/heads/main' in remote)
    after.append(entry)
